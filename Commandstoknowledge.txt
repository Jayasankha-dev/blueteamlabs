Get-ChildItem -Recurse -File | ForEach-Object {
    $dest = Join-Path $PWD $_.Name
    $i = 1
    while (Test-Path $dest) {
        $dest = Join-Path $PWD ($_.BaseName + "_$i" + $_.Extension)
        $i++
    }
    Move-Item $_.FullName -Destination $dest
}
